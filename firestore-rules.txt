rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function isOwner() {
      return signedIn()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == "owner";
    }

    function isStaff() {
      return signedIn()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == "staff";
    }

    function isPOSUser() {
      return isOwner() || isStaff();
    }

    match /users/{userId} {
      allow read: if signedIn() &&
        (request.auth.uid == userId || isOwner());

      allow create, update, delete: if isOwner();
    }

    match /products/{documentId} {
      allow read, create, update: if isPOSUser();
      allow delete: if isOwner();
    }

    match /inventoryUnits/{documentId} {
      allow read, create, update: if isPOSUser();
      allow delete: if isOwner();
    }

    match /sales/{documentId} {
      allow read, create, update: if isPOSUser();
      allow delete: if isOwner();
    }

    match /buys/{documentId} {
      allow read, create, update: if isPOSUser();
      allow delete: if isOwner();
    }

    match /trades/{documentId} {
      allow read, create, update: if isPOSUser();
      allow delete: if isOwner();
    }

    match /repairs/{documentId} {
      allow read, create, update: if isPOSUser();
      allow delete: if isOwner();
    }

    match /refunds/{documentId} {
      allow read, create, update: if isPOSUser();
      allow delete: if isOwner();
    }

    match /exchanges/{documentId} {
      allow read, create, update: if isPOSUser();
      allow delete: if isOwner();
    }

    match /warrantyClaims/{documentId} {
      allow read, create, update: if isPOSUser();
      allow delete: if isOwner();
    }

    match /customers/{documentId} {
      allow read, create, update: if isPOSUser();
      allow delete: if isOwner();
    }

    match /suppliers/{documentId} {
      allow read, create, update: if isPOSUser();
      allow delete: if isOwner();
    }

    match /stockPurchases/{documentId} {
      allow read, create, update: if isPOSUser();
      allow delete: if isOwner();
    }

    match /stockReceipts/{documentId} {
      allow read, create, update: if isPOSUser();
      allow delete: if isOwner();
    }

    match /consignmentAgreements/{documentId} {
      allow read, create, update: if isPOSUser();
      allow delete: if isOwner();
    }

    match /consignmentPayables/{documentId} {
      allow read, create, update: if isPOSUser();
      allow delete: if isOwner();
    }

    match /consignmentSettlements/{documentId} {
      allow read, create, update: if isPOSUser();
      allow delete: if isOwner();
    }

    match /shifts/{documentId} {
      allow read, create, update: if isPOSUser();
      allow delete: if isOwner();
    }

    match /cashMoves/{documentId} {
      allow read, create, update: if isPOSUser();
      allow delete: if isOwner();
    }

    match /counters/{documentId} {
      allow read, create, update: if isPOSUser();
      allow delete: if isOwner();
    }

    match /systemSettings/{documentId} {
      allow read: if isPOSUser();
      allow write: if isOwner();
    }

    match /maintenanceLogs/{documentId} {
      allow read, write: if isOwner();
    }
  }
}
